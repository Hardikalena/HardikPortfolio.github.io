Modular Combat System

A fully scalable and modular approach to sword combat in Unity, designed with SOLID principles and clean architecture. This system demonstrates advanced programming concepts including state machines, strategy patterns, and component-based design.

System Architecture
  • Modular Design: Component-based architecture for easy extension
  • State Machine: Finite state machine for combat states
  • Strategy Pattern: Different combat strategies and styles
  • Observer Pattern: Event-driven combat notifications
  • Scriptable Objects: Data-driven weapon configurations

// Combat State Machine Example
public class CombatStateMachine : MonoBehaviour
{
    public ICombatState CurrentState { get; private set; }
    
    private Dictionary<Type, ICombatState> states;
    
    public void TransitionTo<T>() where T : ICombatState
    {
        CurrentState?.Exit();
        CurrentState = states[typeof(T)];
        CurrentState.Enter();
    }
}

// Weapon Strategy Pattern
public interface IWeaponStrategy
{
    void ExecuteAttack(WeaponData weaponData);
    bool CanExecuteCombo();
}

// Observer Pattern for Combat Events
public static class CombatEvents
{
    public static event Action<float> OnDamageDealt;
    public static event Action<string> OnComboExecuted;
}
